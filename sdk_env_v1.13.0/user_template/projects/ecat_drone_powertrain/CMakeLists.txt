# ecat_drone_powertrain
#   EtherCAT slave application for the ecat_drone_5e3y board (HPM5E3Y)
#
# Build from a *sdk prompt* (start_cmd.cmd) or with the helper script:
#   ..\..\scripts\build.ps1
#
# Manual build:
#   cmake -G Ninja -S . -B build -DBOARD=ecat_drone_5e3y \
#         -DBOARD_SEARCH_PATH=<...>\user_template\boards \
#         -DHPM_BUILD_TYPE=flash_xip -DCMAKE_BUILD_TYPE=debug
#   cmake --build build
#
cmake_minimum_required(VERSION 3.13)

# ---------------------------------------------------------------------------
# EtherCAT EEPROM emulated in flash (same as hpm_sdk/samples/ethercat/ecat_io)
# ---------------------------------------------------------------------------
set(CONFIG_EEPROM_EMULATION 1)

# ---------------------------------------------------------------------------
# CherryUSB device stack (CDC ACM virtual COM port), same configuration as
# hpm_sdk/samples/cherryusb/device/cdc_acm/cdc_acm_vcom
# ---------------------------------------------------------------------------
set(CONFIG_CHERRYUSB 1)
set(CONFIG_USB_DEVICE 1)
set(CONFIG_USB_DEVICE_CDC_ACM 1)

if("${HPM_BUILD_TYPE}" STREQUAL "")
    set(HPM_BUILD_TYPE flash_xip)
endif()

# Output file names: output/ecat_drone_powertrain.elf/.bin/.map
set(APP_NAME ecat_drone_powertrain)

find_package(hpm-sdk REQUIRED HINTS $ENV{HPM_SDK_BASE})

project(ecat_drone_powertrain)

# ---------------------------------------------------------------------------
# Include paths
# ---------------------------------------------------------------------------
sdk_inc(inc)
sdk_inc(.)
sdk_inc(port)
sdk_inc(SSC/Src)
sdk_inc(SSC/StubSrc)
if(CONFIG_EEPROM_EMULATION)
    sdk_compile_definitions(-DCONFIG_EEPROM_EMULATION=1)
endif()

# ---------------------------------------------------------------------------
# EtherCAT slave stack
#   Real SSC code generated by the Beckhoff SSC tool lives in SSC/Src and is
#   detected automatically.  Without it the stub in SSC/StubSrc is built so the
#   project still compiles/links (no EtherCAT communication in that case).
# ---------------------------------------------------------------------------
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/SSC/Src/ecatappl.c)
    message(STATUS "Using Beckhoff SSC slave stack from SSC/Src")
    sdk_compile_definitions(-DAPP_ECAT_SSC_REAL_STACK=1)
    sdk_src(SSC/Src/coeappl.c)
    sdk_src(SSC/Src/ecatappl.c)
    sdk_src(SSC/Src/ecatcoe.c)
    sdk_src(SSC/Src/ecatslv.c)
    sdk_src(SSC/Src/mailbox.c)
    sdk_src(SSC/Src/objdef.c)
    sdk_src(SSC/Src/sdoserv.c)
else()
    message(WARNING
        "SSC/Src/ecatappl.c not found: building with the SSC API stub.\n"
        "  The firmware will compile and run, but EtherCAT communication needs the\n"
        "  real slave stack generated with the Beckhoff SSC Tool - see README_zh.md.")
    sdk_src(SSC/StubSrc/ssc_stub.c)
endif()

# ---------------------------------------------------------------------------
# EtherCAT hardware abstraction layer (copied from hpm_sdk/samples/ethercat/port
# and adapted to the on-chip MII PHY of HPM5E3Y)
# ---------------------------------------------------------------------------
sdk_src(port/hpm_ecat_hw.c)
sdk_src(port/hpm_ecat_phy.c)
if(CONFIG_EEPROM_EMULATION)
    sdk_src(port/hpm_ecat_e2p_emulation.c)
endif()

# ---------------------------------------------------------------------------
# Application
# ---------------------------------------------------------------------------
sdk_app_src(src/main.c)
sdk_app_src(src/app_task.c)
sdk_app_src(src/app_state.c)
sdk_app_src(src/app_log.c)

sdk_app_src(src/ecat_app.c)

# SSC startup glue (same layout as hpm_sdk/samples/ethercat/ecat_io/ecat.c)
sdk_app_src(ecat.c)

sdk_app_src(src/dshot_bidir.c)
sdk_app_src(src/am32_4way.c)
sdk_app_src(src/servo_s288.c)
sdk_app_src(src/led_rgb.c)
sdk_app_src(src/adc_mon.c)
sdk_app_src(src/param.c)
sdk_app_src(src/usb_cdc_app.c)
sdk_app_src(src/host_proto_app.c)

generate_ide_projects()
